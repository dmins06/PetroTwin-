# PetroTwin – Crontab Example for T14 Automated Data Update
# =========================================================================
#
# All cron expressions use Asia/Ho_Chi_Minh local time (ICT, UTC+7).
#
# Requirements:
#   • Linux server (Ubuntu 22.04+ / Debian 12+) with Python 3.11+
#   • Virtual environment with petrotwin installed: /home/petrotwin/.venv
#   • Environment variables in /etc/petrotwin/env (chmod 600)
#   • Kernel file-locking (fcntl.flock) handled natively by PetroTwin Python code
#
# Note on Timezones:
#   Vixie-cron and cronie support CRON_TZ=Asia/Ho_Chi_Minh.
#   If your cron daemon does not support CRON_TZ, either set the system timezone:
#       sudo timedatectl set-timezone Asia/Ho_Chi_Minh
#   or use the equivalent UTC expressions documented in comments below.
#
# Note on Concurrency and Locking:
#   All collectors, runner, and pipeline modules use petrotwin.collectors.locks
#   (kernel fcntl.flock in logs/locks/*.lock). Overlapping executions are safely
#   prevented across cron, manual CLI runs, and subprocesses without relying
#   on separate cron-only wrapper locks.
#
# =========================================================================
# SETUP ON UBUNTU / DEBIAN SERVER
# =========================================================================
#
# 1. Create the system environment file (do NOT commit to version control):
#       sudo mkdir -p /etc/petrotwin
#       sudo tee /etc/petrotwin/env > /dev/null <<'EOF'
#       FRED_API_KEY=your_fred_api_key_here
#       PETROTWIN_DB_HOST=127.0.0.1
#       PETROTWIN_DB_PORT=3306
#       PETROTWIN_DB_USER=petrotwin
#       PETROTWIN_DB_PASSWORD=your_db_password
#       PETROTWIN_DB_NAME=petrotwin
#       EOF
#       sudo chmod 600 /etc/petrotwin/env
#
# 2. Open crontab for service user:
#       sudo -u petrotwin crontab -e
#
# 3. Paste the contents below:
# =========================================================================

# --- Cron Environment Variables ------------------------------------------
SHELL=/bin/bash
BASH_ENV=/etc/petrotwin/env
PATH=/home/petrotwin/.venv/bin:/usr/local/bin:/usr/bin:/bin
CRON_TZ=Asia/Ho_Chi_Minh

# =========================================================================
# JOB 1 — Brent Crude (FRED / EIA)
#   Daily at 06:00 Asia/Ho_Chi_Minh (= 23:00 UTC previous calendar day).
#   Fetches trading-day Brent observations (T+1 lag from international markets).
#   Idempotent: skips weekends/holidays if no new data exists.
#   Cross-process lock: logs/locks/brent.lock.
# =========================================================================
0 6 * * * set -a && . /etc/petrotwin/env && set +a && mkdir -p /home/petrotwin/PetroTwin-/logs/locks && cd /home/petrotwin/PetroTwin- && /home/petrotwin/.venv/bin/python -m petrotwin.collectors.brent >> /home/petrotwin/PetroTwin-/logs/brent.log 2>&1

# =========================================================================
# JOB 2 — USD/VND FX Rate (Algolab, provider_declared SBV upstream)
#   Weekdays (Mon–Fri) at 09:30 Asia/Ho_Chi_Minh (= 02:30 UTC Mon–Fri).
#   SBV central reference rate is announced working mornings (~08:30–09:00).
#   The collector skips weekends automatically.
#   Cross-process lock: logs/locks/fx.lock.
# =========================================================================
30 9 * * 1-5 set -a && . /etc/petrotwin/env && set +a && mkdir -p /home/petrotwin/PetroTwin-/logs/locks && cd /home/petrotwin/PetroTwin- && /home/petrotwin/.venv/bin/python -m petrotwin.collectors.fx >> /home/petrotwin/PetroTwin-/logs/fx.log 2>&1

# =========================================================================
# JOB 3 — Domestic Fuel Price Polling (PVOIL)
#   Thursdays, every 5 minutes from 15:05 to 15:30 Asia/Ho_Chi_Minh (= 08:05–08:30 UTC Thu).
#   MOIT/PVOIL release retail fuel adjustment notices on Thursdays at ~15:00 ICT.
#   The collector is idempotent: stops appending once the new adjustment date exists.
#   Cross-process lock: logs/locks/domestic_fuel.lock.
# =========================================================================
5-30/5 15 * * 4 set -a && . /etc/petrotwin/env && set +a && mkdir -p /home/petrotwin/PetroTwin-/logs/locks && cd /home/petrotwin/PetroTwin- && /home/petrotwin/.venv/bin/python -m petrotwin.collectors.domestic_fuel >> /home/petrotwin/PetroTwin-/logs/domestic_fuel.log 2>&1

# =========================================================================
# JOB 4a — Pipeline & Loader (Non-Thursday: Sun, Mon, Tue, Wed, Fri, Sat)
#   Daily at 10:00 Asia/Ho_Chi_Minh (= 03:00 UTC).
#   Runs pipeline validation and MySQL DB loader without re-collecting sources.
#   Sources (Brent at 06:00, FX at 09:30) have already executed in their own slots.
#   Cross-process locks: logs/locks/runner.lock and logs/locks/pipeline.lock.
# =========================================================================
0 10 * * 0,1,2,3,5,6 set -a && . /etc/petrotwin/env && set +a && mkdir -p /home/petrotwin/PetroTwin-/logs/locks && cd /home/petrotwin/PetroTwin- && /home/petrotwin/.venv/bin/python -m petrotwin.collectors.runner --pipeline-only >> /home/petrotwin/PetroTwin-/logs/runner.log 2>&1

# =========================================================================
# JOB 4b — Pipeline & Loader (Thursday adjustment day)
#   Thursdays at 16:35 Asia/Ho_Chi_Minh (= 09:35 UTC Thu).
#   Runs safely after the 15:00–15:30 PVOIL polling window completes.
#   Integrates fuel price notice into master dataset and pushes to MySQL.
#   Cross-process locks: logs/locks/runner.lock and logs/locks/pipeline.lock.
# =========================================================================
35 16 * * 4 set -a && . /etc/petrotwin/env && set +a && mkdir -p /home/petrotwin/PetroTwin-/logs/locks && cd /home/petrotwin/PetroTwin- && /home/petrotwin/.venv/bin/python -m petrotwin.collectors.runner --pipeline-only >> /home/petrotwin/PetroTwin-/logs/runner.log 2>&1

# =========================================================================
# LOG ROTATION CONFIGURATION (/etc/logrotate.d/petrotwin)
# =========================================================================
# /home/petrotwin/PetroTwin-/logs/*.log {
#     daily
#     rotate 30
#     compress
#     missingok
#     notifempty
#     create 640 petrotwin petrotwin
# }
